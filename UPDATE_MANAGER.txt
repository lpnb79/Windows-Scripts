@echo off
setlocal enabledelayedexpansion

:: Set log file path (saves to the same folder as the script)
set "LOG_FILE=%~dp0winget_update_log.txt"

echo ===================================================
echo   Windows 11 Package Update Manager
echo ===================================================
echo.

:: 1. Display available updates
echo Checking for available updates...
echo.
winget upgrade
echo.

:: 2. Prompt user for action
echo Choose an option:
echo [1] Update EVERYTHING automatically
echo [2] Update a SPECIFIC program
echo [3] Exit without updating
echo.
set /p "CHOICE=Enter choice (1, 2, or 3): "

if "%CHOICE%"=="1" (
    echo.
    echo Updating all programs...
    echo [%date% %time%] STARTED: Bulk upgrade all apps >> "%LOG_FILE%"

    winget upgrade --all --include-unknown 

    echo [%date% %time%] FINISHED: Bulk upgrade process completed >> "%LOG_FILE%"
    echo.
    echo All updates completed. Start and finish logged to: %LOG_FILE%
) ^
else if "%CHOICE%"=="2" (
    echo.
    set /p "APP_ID=Enter the exact ID or Name of the app to update: "
    echo.
    echo Updating %APP_ID%...
    echo [%date% %time%] STARTED: Upgrading %APP_ID% >> "%LOG_FILE%"
    winget upgrade "%APP_ID%"
    echo [%date% %time%] FINISHED: Upgrade process for %APP_ID% completed >> "%LOG_FILE%"
    echo.
    echo Update completed. Results logged to: %LOG_FILE%
) ^
else (
    echo.
    echo Exiting. No changes made.
)

echo.
pause
